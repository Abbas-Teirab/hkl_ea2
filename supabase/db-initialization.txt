--//--------------------------------------------- 
--nodes table

CREATE TABLE nodes(
	id uuid DEFAULT uuid_generate_v4() PRIMARY KEY,
	name TEXT,
	location TEXT,
	transmission_period FLOAT DEFAULT 60,
	enabled BOOLEAN DEFAULT TRUE,
	created_at timestamptz DEFAULT now()
);

--//--------------------------------------------- 
--sensors table

CREATE TABLE sensors(
	id uuid DEFAULT uuid_generate_v4() PRIMARY KEY,
	name TEXT,
	mac_address TEXT,
	ip_address TEXT,
	locked BOOLEAN DEFAULT TRUE,
	temperature FLOAT,
	pressure FLOAT,
	humidity FLOAT,
	altitude FLOAT,
	created_at timestamptz DEFAULT now()
);

alter publication supabase_realtime add table sensors;

--//--------------------------------------------- 
--locks table

CREATE TABLE locks(
	id uuid DEFAULT uuid_generate_v4() PRIMARY KEY,
	name TEXT,
	mac_address TEXT,
	ip_address TEXT,
	locked BOOLEAN DEFAULT TRUE,
	created_at timestamptz DEFAULT now()
);

alter publication supabase_realtime add table locks;
